The page sends the class name followed by a new line, for example Class 1. Match your own class names in the code below.

USB: MakeCode JavaScript 
let cmd = ""
serial.setBaudRate(BaudRate.BaudRate115200)
serial.onDataReceived(serial.delimiters(Delimiters.NewLine), function () {
    cmd = serial.readLine().trim()
    if (cmd == "Class 1") {
        basic.showIcon(IconNames.Happy)
    } else if (cmd == "Class 2") {
        basic.showIcon(IconNames.Sad)
    } else {
        basic.clearScreen()
    }
})
Bluetooth: MakeCode JavaScript (add the Bluetooth extension, turn off "No Pairing Required") 
bluetooth.startUartService()
bluetooth.onUartDataReceived(serial.delimiters(Delimiters.NewLine), function () {
    let cmd = bluetooth.uartReadUntil(serial.delimiters(Delimiters.NewLine)).trim()
    if (cmd == "Class 1") {
        basic.showIcon(IconNames.Happy)
    } else if (cmd == "Class 2") {
        basic.showIcon(IconNames.Sad)
    } else {
        basic.clearScreen()
    }
})
